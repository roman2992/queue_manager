from datetime import datetime, timedelta, timezone
from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel, Field
from sqlalchemy import select, func
from sqlalchemy.ext.asyncio import AsyncSession
from app.api.dependencies import current_user, require_admin
from app.core.database import get_db
from app.core.security import hash_password, verify_password, create_access_token
from app.models import User, Business, Service, Queue, Ticket, TicketStatus, Role

router=APIRouter()
class AuthIn(BaseModel): phone:str; password:str=Field(min_length=6); role:Role=Role.CUSTOMER
class BusinessIn(BaseModel): name:str; address:str|None=None; latitude:float|None=None; longitude:float|None=None
class ServiceIn(BaseModel): name:str; avg_duration_minutes:int=30
class TicketOut(BaseModel): id:int; queue_sequence:int; status:str; people_ahead:int; expires_at:datetime|None

@router.post("/auth/register")
async def register(data:AuthIn, db:AsyncSession=Depends(get_db)):
    if await db.scalar(select(User).where(User.phone==data.phone)): raise HTTPException(409,"Phone already registered")
    user=User(phone=data.phone,password_hash=hash_password(data.password),role=data.role.value); db.add(user); await db.commit(); await db.refresh(user)
    return {"access_token":create_access_token(str(user.id),user.role),"user_id":user.id,"role":user.role}

@router.post("/auth/login")
async def login(data:AuthIn, db:AsyncSession=Depends(get_db)):
    user=await db.scalar(select(User).where(User.phone==data.phone))
    if not user or not verify_password(data.password,user.password_hash): raise HTTPException(401,"Invalid credentials")
    return {"access_token":create_access_token(str(user.id),user.role),"user_id":user.id,"role":user.role}

@router.post("/admin/businesses")
async def create_business(data:BusinessIn,user=Depends(require_admin),db:AsyncSession=Depends(get_db)):
    b=Business(owner_id=user.id,**data.model_dump()); db.add(b); await db.commit(); await db.refresh(b); return b

@router.post("/admin/businesses/{business_id}/services")
async def create_service(business_id:int,data:ServiceIn,user=Depends(require_admin),db:AsyncSession=Depends(get_db)):
    b=await db.scalar(select(Business).where(Business.id==business_id,Business.owner_id==user.id))
    if not b: raise HTTPException(404,"Business not found")
    s=Service(business_id=business_id,**data.model_dump()); db.add(s); await db.commit(); await db.refresh(s)
    q=Queue(service_id=s.id,name=s.name); db.add(q); await db.commit(); await db.refresh(q)
    return {"service":s,"queue":q}

@router.get("/businesses")
async def businesses(db:AsyncSession=Depends(get_db)): return (await db.scalars(select(Business))).all()

@router.post("/queues/{queue_id}/tickets",response_model=TicketOut)
async def join_queue(queue_id:int,user=Depends(current_user),db:AsyncSession=Depends(get_db)):
    q=await db.scalar(select(Queue).where(Queue.id==queue_id).with_for_update())
    if not q or not q.is_open: raise HTTPException(400,"Queue is closed")
    existing=await db.scalar(select(Ticket).where(Ticket.queue_id==queue_id,Ticket.user_id==user.id,Ticket.status.in_([s.value for s in TicketStatus if s in {TicketStatus.WAITING,TicketStatus.CHECKED_IN,TicketStatus.CALLED,TicketStatus.IN_SERVICE}])))
    if existing: raise HTTPException(409,"Already in queue")
    seq=q.next_sequence; q.next_sequence+=1
    t = Ticket(
        queue_id=queue_id,
        user_id=user.id,
        queue_sequence=seq,
        expires_at=None,
    )
    db.add(t)
    await db.commit()
    await db.refresh(t)
    ahead=await db.scalar(select(func.count()).select_from(Ticket).where(Ticket.queue_id==queue_id,Ticket.queue_sequence<seq,Ticket.status.in_([s.value for s in [TicketStatus.WAITING,TicketStatus.CHECKED_IN,TicketStatus.CALLED,TicketStatus.IN_SERVICE]])))
    return TicketOut(id=t.id,queue_sequence=seq,status=t.status,people_ahead=ahead or 0,expires_at=t.expires_at)

@router.get("/tickets/{ticket_id}",response_model=TicketOut)
async def ticket(ticket_id:int,user=Depends(current_user),db:AsyncSession=Depends(get_db)):
    t=await db.scalar(select(Ticket).where(Ticket.id==ticket_id,Ticket.user_id==user.id))
    if not t: raise HTTPException(404,"Ticket not found")
    ahead=await db.scalar(select(func.count()).select_from(Ticket).where(Ticket.queue_id==t.queue_id,Ticket.queue_sequence<t.queue_sequence,Ticket.status.in_([x.value for x in [TicketStatus.WAITING,TicketStatus.CHECKED_IN,TicketStatus.CALLED,TicketStatus.IN_SERVICE]])))
    return TicketOut(id=t.id,queue_sequence=t.queue_sequence,status=t.status,people_ahead=ahead or 0,expires_at=t.expires_at)

@router.post("/tickets/{ticket_id}/check-in",response_model=TicketOut)
async def check_in(ticket_id:int,user=Depends(current_user),db:AsyncSession=Depends(get_db)):
    t=await db.scalar(select(Ticket).where(Ticket.id==ticket_id,Ticket.user_id==user.id).with_for_update())
    if not t: raise HTTPException(404,"Ticket not found")
    if t.status != TicketStatus.WAITING.value: raise HTTPException(400,"Ticket cannot be checked in")
    t.status=TicketStatus.CHECKED_IN.value; t.check_in_at=datetime.now(timezone.utc); await db.commit(); await db.refresh(t)
    ahead=await db.scalar(select(func.count()).select_from(Ticket).where(Ticket.queue_id==t.queue_id,Ticket.queue_sequence<t.queue_sequence,Ticket.status.in_([x.value for x in [TicketStatus.WAITING,TicketStatus.CHECKED_IN,TicketStatus.CALLED,TicketStatus.IN_SERVICE]])))
    return TicketOut(id=t.id,queue_sequence=t.queue_sequence,status=t.status,people_ahead=ahead or 0,expires_at=t.expires_at)

async def _owned_queue(queue_id:int,user,db):
    q=await db.scalar(select(Queue).join(Service,Queue.service_id==Service.id).join(Business,Business.id==Service.business_id).where(Queue.id==queue_id,Business.owner_id==user.id))
    if not q: raise HTTPException(404,"Queue not found")
    return q

@router.get("/admin/queues/{queue_id}")
async def admin_queue(queue_id:int,user=Depends(require_admin),db:AsyncSession=Depends(get_db)):
    q=await _owned_queue(queue_id,user,db)
    rows=(await db.scalars(select(Ticket).where(Ticket.queue_id==q.id).order_by(Ticket.queue_sequence))).all()
    return [{"id":t.id,"sequence":t.queue_sequence,"status":t.status,"expires_at":t.expires_at} for t in rows]

@router.post("/admin/queues/{queue_id}/call-next")
async def call_next(queue_id:int,user=Depends(require_admin),db:AsyncSession=Depends(get_db)):
    q=await _owned_queue(queue_id,user,db)
    t=await db.scalar(select(Ticket).where(Ticket.queue_id==q.id,Ticket.status.in_([TicketStatus.WAITING.value,TicketStatus.CHECKED_IN.value])).order_by(Ticket.queue_sequence).with_for_update(skip_locked=True))
    if not t: raise HTTPException(409,"No waiting tickets")
    # t.status=TicketStatus.CALLED.value; t.called_at=datetime.now(timezone.utc); await db.commit(); return {"ticket_id":t.id,"sequence":t.queue_sequence,"status":t.status}
    now = datetime.now(timezone.utc)
    t.status = TicketStatus.CALLED.value
    t.called_at = now
    t.expires_at = now + timedelta(hours=2)

    await db.commit()

    return {
        "ticket_id": t.id,
        "sequence": t.queue_sequence,
        "status": t.status,
    }


@router.post("/admin/tickets/{ticket_id}/start")
async def start_ticket(ticket_id:int,user=Depends(require_admin),db:AsyncSession=Depends(get_db)):
    t=await db.scalar(select(Ticket).join(Queue,Ticket.queue_id==Queue.id).join(Service,Queue.service_id==Service.id).join(Business,Business.id==Service.business_id).where(Ticket.id==ticket_id,Business.owner_id==user.id).with_for_update())
    if not t or t.status != TicketStatus.CALLED.value: raise HTTPException(400,"Ticket cannot be started")
    t.status=TicketStatus.IN_SERVICE.value; t.service_started_at=datetime.now(timezone.utc); await db.commit(); return {"ticket_id":t.id,"status":t.status}

@router.post("/admin/tickets/{ticket_id}/complete")
async def complete_ticket(ticket_id:int,user=Depends(require_admin),db:AsyncSession=Depends(get_db)):
    t=await db.scalar(select(Ticket).join(Queue,Ticket.queue_id==Queue.id).join(Service,Queue.service_id==Service.id).join(Business,Business.id==Service.business_id).where(Ticket.id==ticket_id,Business.owner_id==user.id).with_for_update())
    if not t or t.status != TicketStatus.IN_SERVICE.value: raise HTTPException(400,"Ticket cannot be completed")
    now=datetime.now(timezone.utc); t.status=TicketStatus.COMPLETED.value; t.completed_at=now; t.resolved_at=now; t.resolved_reason="COMPLETED"; await db.commit(); return {"ticket_id":t.id,"status":t.status,"resolved_at":t.resolved_at}

from datetime import datetime
from enum import Enum
from sqlalchemy import Boolean, DateTime, ForeignKey, Integer, String, Text, func
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column, relationship

class Base(DeclarativeBase): pass
class Role(str, Enum): CUSTOMER="CUSTOMER"; ADMIN="ADMIN"
class TicketStatus(str, Enum): WAITING="WAITING"; CHECKED_IN="CHECKED_IN"; CALLED="CALLED"; IN_SERVICE="IN_SERVICE"; COMPLETED="COMPLETED"; EXPIRED="EXPIRED"; CANCELLED="CANCELLED"

class User(Base):
    __tablename__="users"
    id: Mapped[int]=mapped_column(primary_key=True)
    phone: Mapped[str]=mapped_column(String(32), unique=True, index=True)
    password_hash: Mapped[str]=mapped_column(Text())
    role: Mapped[str]=mapped_column(String(16), default=Role.CUSTOMER.value)
    created_at: Mapped[datetime]=mapped_column(DateTime(timezone=True), server_default=func.now())

class Business(Base):
    __tablename__="businesses"
    id: Mapped[int]=mapped_column(primary_key=True)
    owner_id: Mapped[int]=mapped_column(ForeignKey("users.id"), index=True)
    name: Mapped[str]=mapped_column(String(160))
    address: Mapped[str|None]=mapped_column(String(300), nullable=True)
    latitude: Mapped[float|None]=mapped_column(nullable=True)
    longitude: Mapped[float|None]=mapped_column(nullable=True)
    is_open: Mapped[bool]=mapped_column(Boolean, default=True)

class Service(Base):
    __tablename__="services"
    id: Mapped[int]=mapped_column(primary_key=True)
    business_id: Mapped[int]=mapped_column(ForeignKey("businesses.id"), index=True)
    name: Mapped[str]=mapped_column(String(160))
    avg_duration_minutes: Mapped[int]=mapped_column(Integer, default=30)
    active: Mapped[bool]=mapped_column(Boolean, default=True)

class Queue(Base):
    __tablename__="queues"
    id: Mapped[int]=mapped_column(primary_key=True)
    service_id: Mapped[int]=mapped_column(ForeignKey("services.id"), index=True)
    name: Mapped[str]=mapped_column(String(160))
    is_open: Mapped[bool]=mapped_column(Boolean, default=True)
    next_sequence: Mapped[int]=mapped_column(Integer, default=1)

class Ticket(Base):
    __tablename__="queue_tickets"
    id: Mapped[int]=mapped_column(primary_key=True)
    queue_id: Mapped[int]=mapped_column(ForeignKey("queues.id"), index=True)
    user_id: Mapped[int]=mapped_column(ForeignKey("users.id"), index=True)
    queue_sequence: Mapped[int]=mapped_column(Integer)
    status: Mapped[str]=mapped_column(String(20), default=TicketStatus.WAITING.value, index=True)
    created_at: Mapped[datetime]=mapped_column(DateTime(timezone=True), server_default=func.now())
    check_in_at: Mapped[datetime|None]=mapped_column(DateTime(timezone=True), nullable=True)
    called_at: Mapped[datetime|None]=mapped_column(DateTime(timezone=True), nullable=True)
    service_started_at: Mapped[datetime|None]=mapped_column(DateTime(timezone=True), nullable=True)
    completed_at: Mapped[datetime|None]=mapped_column(DateTime(timezone=True), nullable=True)
    expires_at: Mapped[datetime|None]=mapped_column(DateTime(timezone=True), nullable=True)
    resolved_at: Mapped[datetime|None]=mapped_column(DateTime(timezone=True), nullable=True)
    resolved_reason: Mapped[str|None]=mapped_column(String(32), nullable=True)


========== apps/api/app/workers/ticket_expiration.py ==========

from datetime import datetime, timezone

from sqlalchemy import select

from app.core.database import SessionLocal
from app.models import Ticket, TicketStatus


async def expire_due_tickets():
    async with SessionLocal() as db:
        now = datetime.now(timezone.utc)

        rows = (
            await db.scalars(
                select(Ticket).where(
                    Ticket.expires_at <= now,
                    Ticket.status == TicketStatus.CALLED.value,
                )
            )
        ).all()

        for ticket in rows:
            ticket.status = TicketStatus.EXPIRED.value
            ticket.resolved_at = now
            ticket.resolved_reason = "TTL"

        await db.commit()

        return len(rows)

import asyncio

from celery import Celery

from app.core.config import settings

celery_app = Celery(
    "queue_manager",
    broker=settings.redis_url,
    backend=settings.redis_url,
)


@celery_app.task
def expire_due_tickets_task():
    from app.workers.ticket_expiration import expire_due_tickets

    return asyncio.run(expire_due_tickets())


celery_app.conf.beat_schedule = {
    "expire-due-tickets-every-minute": {
        "task": "app.workers.celery_app.expire_due_tickets_task",
        "schedule": 60.0,
    },
}

celery_app.conf.timezone = "UTC"
celery_app.conf.enable_utc = True

from sqlalchemy.ext.asyncio import (
    AsyncSession,
    async_sessionmaker,
    create_async_engine,
)
from sqlalchemy.pool import NullPool

from app.core.config import settings

engine = create_async_engine(
    settings.database_url,
    pool_pre_ping=True,
    poolclass=NullPool,
)

SessionLocal = async_sessionmaker(
    engine,
    class_=AsyncSession,
    expire_on_commit=False,
)

async def get_db():
    async with SessionLocal() as session:
        yield session
